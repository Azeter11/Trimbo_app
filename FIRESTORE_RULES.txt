rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ─── 1. ATURAN USERS & INTEGRASI (DIBUKA UNTUK ACADEMIC ATELIER) ───
    
    // Ini buat koleksi users umum
    match /users/{userId} {
      allow read, write: if true;
    }

    // ✅ INI KUNCINYA: Biar data dari Dashboard lo bisa masuk tanpa error merah!
    match /trimbo_users/{userId} {
      allow read, write: if true;
    }
    
    // ─── 2. ATURAN BAWAAN TRIMBO (TIDAK DISENTUH & TETAP AMAN) ───
    
    // Kelas: guru bisa buat, semua authenticated user bisa baca
    match /classes/{classId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }
    
    // Tugas: semua authenticated user bisa baca
    match /assignments/{assignmentId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }
    
    // ─── 3. ATURAN QUESTIONS (DIBACA TAMU, DITULIS ADMIN/GURU TRIMBO) ───
    match /questions/{questionId} {
      allow read: if true;                 // Biar soal langsung muncul di HP lo
      allow write: if request.auth != null; // Nulis soal tetep harus login
    }
    
    // ─── 4. ATURAN SUBMISSIONS (DITULIS TAMU, DIBACA ADMIN/GURU TRIMBO) ───
    match /submissions/{submissionId} {
      allow write: if true;                // Biar skor dari Academic Atelier bisa tersimpan
      allow read: if request.auth != null; // Rekap nilai tetep privat buat guru
    }
  }
}
