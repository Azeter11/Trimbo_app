// Firebase Storage Security Rules untuk Trimbo App
// Copy paste aturan ini ke Firebase Console > Storage > Rules

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    
    // ===== ATURAN UNTUK ESSAY SUBMISSIONS (PDF) =====
    match /essay_submissions/{fileName} {
      // Hanya user yang sudah login yang bisa upload
      allow write: if request.auth != null 
                   && resource == null // File belum ada (upload baru)
                   && request.resource.contentType == 'application/pdf'
                   && request.resource.size <= 10 * 1024 * 1024; // Maksimal 10MB
      
      // User yang sudah login bisa baca file
      allow read: if request.auth != null;
    }
    
    // ===== ATURAN UNTUK QUESTION IMAGES (Gambar Soal) =====
    match /question_images/{fileName} {
      // Hanya user yang sudah login yang bisa upload
      allow write: if request.auth != null 
                   && resource == null // File belum ada (upload baru)
                   && request.resource.contentType.matches('image/.*')
                   && request.resource.size <= 5 * 1024 * 1024; // Maksimal 5MB
      
      // Semua user yang login bisa baca gambar soal
      allow read: if request.auth != null;
    }
    
    // ===== ATURAN UNTUK WORD DOCUMENTS (Generate Soal) =====
    match /word_documents/{fileName} {
      // Hanya guru/dosen yang bisa upload Word document
      allow write: if request.auth != null 
                   && resource == null // File belum ada (upload baru)
                   && (request.resource.contentType == 'application/vnd.openxmlformats-officedocument.wordprocessingml.document'
                       || request.resource.contentType == 'application/msword')
                   && request.resource.size <= 20 * 1024 * 1024; // Maksimal 20MB
      
      // Hanya uploader yang bisa baca file Word
      allow read: if request.auth != null;
    }
    
    // ===== ATURAN UNTUK PROFILE IMAGES =====
    match /profile_images/{fileName} {
      // User bisa upload foto profil sendiri
      allow write: if request.auth != null 
                   && resource == null // File belum ada (upload baru)
                   && request.resource.contentType.matches('image/.*')
                   && request.resource.size <= 5 * 1024 * 1024; // Maksimal 5MB
      
      // Semua user yang login bisa lihat foto profil
      allow read: if request.auth != null;
    }
    
    // ===== ATURAN DEFAULT (DENY SEMUA) =====
    // File di lokasi lain tidak boleh diakses
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
